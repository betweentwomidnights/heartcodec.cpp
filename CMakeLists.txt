cmake_minimum_required(VERSION 3.20)
project(heartcodec_cpp VERSION 0.2.0 LANGUAGES C CXX)
set(heartcodec_top_level OFF)
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
    set(heartcodec_top_level ON)
endif()
include(CTest)
if(WIN32)
    include(ExternalProject)
    set_property(DIRECTORY PROPERTY EP_BASE "${CMAKE_BINARY_DIR}/ep")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(HEARTCODEC_CUDA "Build CUDA backend" OFF)
option(HEARTCODEC_VULKAN "Build Vulkan backend" OFF)
option(HEARTCODEC_METAL "Build Metal backend" OFF)
option(HEARTCODEC_STATIC "Build a static native codec library" ON)
option(HEARTCODEC_BUILD_TOOLS "Build CLI tools" ${heartcodec_top_level})
option(HEARTCODEC_BUILD_TESTS "Build native tests" ${heartcodec_top_level})
set(HEARTCODEC_GGML_TARGET "" CACHE STRING "Existing shared-fork GGML target, e.g. ggml or ggml::ggml")
if(NOT HEARTCODEC_GGML_TARGET)
    if(TARGET ggml)
        set(HEARTCODEC_GGML_TARGET ggml)
    elseif(TARGET ggml::ggml)
        set(HEARTCODEC_GGML_TARGET ggml::ggml)
    else()
        set(BUILD_SHARED_LIBS OFF CACHE BOOL "Build shared dependencies")
        set(GGML_BUILD_TESTS OFF CACHE BOOL "Build dependency tests")
        set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "Build dependency examples")
        foreach(backend CUDA VULKAN METAL)
            set(GGML_${backend} ${HEARTCODEC_${backend}} CACHE BOOL "GGML backend" FORCE)
        endforeach()
        if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/ggml/CMakeLists.txt")
            message(FATAL_ERROR "Run git submodule update --init --recursive before configuring")
        endif()
        add_subdirectory(ggml)
        set(HEARTCODEC_GGML_TARGET ggml)
    endif()
endif()
if(NOT TARGET ${HEARTCODEC_GGML_TARGET})
    message(FATAL_ERROR "HEARTCODEC_GGML_TARGET must name an existing GGML target")
endif()
set(heartcodec_sources src/heartcodec.cpp src/heartcodec-model.cpp src/heartcodec-sq.cpp src/heartcodec-flow.cpp src/heartcodec-api.cpp src/heartcodec-rng.cpp src/heartcodec-encoder-model.cpp src/heartcodec-encoder-query.cpp src/heartcodec-encoder-audio.cpp src/heartcodec-encoder-features.cpp src/heartcodec-encoder.cpp)
if(HEARTCODEC_CUDA)
    enable_language(CUDA)
    find_package(CUDAToolkit REQUIRED)
    list(APPEND heartcodec_sources src/heartcodec-rng.cu)
endif()
if(HEARTCODEC_STATIC)
    add_library(heartcodec STATIC ${heartcodec_sources})
else()
    add_library(heartcodec SHARED ${heartcodec_sources})
    target_compile_definitions(heartcodec PUBLIC HEARTCODEC_SHARED PRIVATE HEARTCODEC_BUILD)
endif()
add_library(heartcodec::heartcodec ALIAS heartcodec)
add_library(heartcodec::ggml_dependency INTERFACE IMPORTED)
set_target_properties(heartcodec::ggml_dependency PROPERTIES INTERFACE_LINK_LIBRARIES "ggml::ggml")
target_include_directories(heartcodec PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:include> PRIVATE src)
target_link_libraries(heartcodec PRIVATE $<BUILD_INTERFACE:${HEARTCODEC_GGML_TARGET}> $<INSTALL_INTERFACE:heartcodec::ggml_dependency>)
target_compile_features(heartcodec PUBLIC cxx_std_17)
if(HEARTCODEC_CUDA)
    target_compile_definitions(heartcodec PRIVATE HEARTCODEC_RNG_CUDA)
    target_link_libraries(heartcodec PRIVATE CUDA::cudart)
    set_target_properties(heartcodec PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
endif()
if(NOT MSVC)
    target_compile_options(heartcodec PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-ffp-contract=off>)
endif()
set_target_properties(heartcodec PROPERTIES POSITION_INDEPENDENT_CODE ON)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
install(TARGETS heartcodec EXPORT heartcodec-targets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES include/heartcodec.h include/heartcodec_rng.h include/heartcodec_encoder.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES THIRD_PARTY_NOTICES.md DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/heartcodec)
install(FILES third_party/LICENSE-Apache-2.0.txt third_party/LICENSE-MusicFM.txt
    DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/heartcodec/third_party)
configure_package_config_file(cmake/heartcodec-config.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/heartcodec-config.cmake
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/heartcodec)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/heartcodec-config-version.cmake
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(EXPORT heartcodec-targets NAMESPACE heartcodec:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/heartcodec)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/heartcodec-config.cmake ${CMAKE_CURRENT_BINARY_DIR}/heartcodec-config-version.cmake
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/heartcodec)
if(HEARTCODEC_BUILD_TOOLS)
    add_executable(heartcodec-encode tools/heartcodec-encode.cpp)
    target_link_libraries(heartcodec-encode PRIVATE heartcodec)
    if(HEARTCODEC_STATIC)
        target_compile_definitions(heartcodec-encode PRIVATE HEARTCODEC_ENCODER_DIAGNOSTICS)
        target_include_directories(heartcodec-encode PRIVATE src)
        target_link_libraries(heartcodec-encode PRIVATE ${HEARTCODEC_GGML_TARGET})
    endif()
    add_executable(heartcodec-cli tools/heartcodec-cli.cpp)
    target_link_libraries(heartcodec-cli PRIVATE heartcodec)
    add_executable(heartcodec-quantize tools/heartcodec-quantize.cpp)
    target_link_libraries(heartcodec-quantize PRIVATE ${HEARTCODEC_GGML_TARGET})
    add_executable(heartcodec-flow tools/heartcodec-flow.cpp)
    target_link_libraries(heartcodec-flow PRIVATE heartcodec)
    add_executable(heartcodec-rng tools/heartcodec-rng.cpp)
    target_link_libraries(heartcodec-rng PRIVATE heartcodec)
endif()
if(BUILD_TESTING AND HEARTCODEC_BUILD_TESTS)
    add_executable(test-encoder-api tests/test-encoder-api.c)
    target_link_libraries(test-encoder-api PRIVATE heartcodec)
    if(MSVC)
        target_compile_options(test-encoder-api PRIVATE /UNDEBUG)
    else()
        target_compile_options(test-encoder-api PRIVATE -UNDEBUG)
    endif()
    add_test(NAME encoder-api COMMAND test-encoder-api)
    add_executable(test-rng tests/test-rng.cpp)
    target_link_libraries(test-rng PRIVATE heartcodec)
    add_test(NAME rng-api COMMAND test-rng)
    add_executable(test-matmul-source-f32 tests/test-matmul-source-f32.cpp)
    target_link_libraries(test-matmul-source-f32 PRIVATE ${HEARTCODEC_GGML_TARGET})
    add_test(NAME matmul-source-f32 COMMAND test-matmul-source-f32)
    set_tests_properties(matmul-source-f32 PROPERTIES SKIP_RETURN_CODE 77
        ENVIRONMENT "GGML_CUDA_TF32=0;GGML_CUDA_CUBLAS_COMPUTE_TYPE=auto")
    add_executable(test-sq-quantize tests/test-sq-quantize.cpp)
    target_link_libraries(test-sq-quantize PRIVATE heartcodec)
    add_test(NAME sq-quantize COMMAND test-sq-quantize)
    add_executable(test-c-api tests/test-c-api.c)
    target_link_libraries(test-c-api PRIVATE heartcodec)
    add_test(NAME c-api COMMAND test-c-api)
endif()
