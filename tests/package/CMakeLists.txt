cmake_minimum_required(VERSION 3.20)
project(heartcodec_consumer LANGUAGES C CXX)
if(HEARTCODEC_SOURCE_DIR)
    # A downstream application may already own the shared GGML dependency.
    if(GGML_SOURCE_DIR)
        add_subdirectory("${GGML_SOURCE_DIR}" ggml)
    else()
        find_package(ggml CONFIG REQUIRED)
    endif()
    set(HEARTCODEC_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
    set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
    add_subdirectory("${HEARTCODEC_SOURCE_DIR}" heartcodec)
else()
    find_package(heartcodec CONFIG REQUIRED)
endif()
add_executable(consumer main.c)
target_link_libraries(consumer PRIVATE heartcodec::heartcodec)
